## Executive Summary

This specification defines the implementation of a **modular, reusable AI Analysis system** that can be deployed across all top-level report pages in the Midas application. The system will provide contextual financial insights using LLM-powered analysis while adhering to DRY, KISS, and best coding practices.

### Key Principles

- ✅ **Simple & Modular:** Reusable component with minimal configuration
- ✅ **DRY (Don't Repeat Yourself):** Shared logic, single source of truth
- ✅ **KISS (Keep It Simple):** No over-engineering, clear abstractions
- ✅ **Type-Safe:** Full TypeScript coverage
- ✅ **Extensible:** Easy to add new page types

### Success Criteria

1. AI Analysis deployed on 6+ pages with <50 lines of code per integration
2. Shared component handles all analysis logic
3. Config-driven behavior (no hardcoded logic per page)
4. <100ms overhead on page load (lazy/background generation)
5. 15-min cache hit rate >70%

## Goals & Non-Goals

### Goals ✅

- Deploy AI Analysis to all top-level sidebar pages (P&L, BS, CF, Sales, Expenses, Journal)
- Unified component architecture (single source of truth)
- Config-driven page-specific behavior
- Template-based prompt system
- Historical data awareness (analyze old data correctly)
- Simple rate limiting (prevent abuse)
- Maintain existing 3-section format (Strategic Insights, Forward-Looking, Prioritized Actions)
- 15-min TTL caching (configurable per page)
- Props-based component API
- Error handling with retry capability

### Non-Goals ❌

- Multiple analysis structures (keep 3-section format)
- Real-time/streaming analysis
- Custom user preferences (future enhancement)
- Export functionality (future enhancement)
- AI choosing sections dynamically (keep fixed structure)
- Sub-page analysis (customer detail, vendor detail)
- Multi-language support (future)
- Historical analysis comparison (future)

- Calls API endpoint
- Handles errors gracefully
- Updates state

- Parses markdown sections
- Applies styling (reused from existing implementation)
- Makes items clickable for chat integration
- Returns sanitized HTML

#### Component Code (Simplified)

```typescript
'use client'

import { useState, useEffect, useRef } from 'react'
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Sparkles, Loader2, RefreshCw, AlertCircle } from 'lucide-react'
import { AI_ANALYSIS_CONFIG } from './ai-analysis.config'
import { formatAnalysis } from './formatters'
import DOMPurify from 'isomorphic-dompurify'
import type { AIAnalysisCardProps } from './types'

export function AIAnalysisCard({
  pageType,
  data,
  dateRange,
  context,
  autoGenerate = true,
  className
}: AIAnalysisCardProps) {
  // State
  const [analysis, setAnalysis] = useState<string | null>(null)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const [hasGenerated, setHasGenerated] = useState(false)
  const [remainingCooldown, setRemainingCooldown] = useState(0)

  // Refs
  const hasAutoGeneratedRef = useRef(false)
  const lastForceRegenTimeRef = useRef<number>(0)
  const FORCE_REGEN_COOLDOWN_MS = 30000

  // Get config for this page type
  const config = AI_ANALYSIS_CONFIG[pageType]
  if (!config) {
    console.error(`No AI analysis config found for pageType: ${pageType}`)
    return null
  }

  // Auto-generation effect
  useEffect(() => {
    if (autoGenerate && !hasAutoGeneratedRef.current && !hasGenerated && !loading) {
      hasAutoGeneratedRef.current = true
      generateAnalysis(false)
    }
  }, [autoGenerate, hasGenerated, loading])

  // Cooldown timer effect
  useEffect(() => {
    if (remainingCooldown <= 0) return
    const timer = setInterval(() => {
      setRemainingCooldown((prev) => Math.max(0, prev - 1))
    }, 1000)
    return () => clearInterval(timer)
  }, [remainingCooldown])

  // Generate analysis
  const generateAnalysis = async (forceRegenerate: boolean = false) => {
    // Check cooldown for force regeneration
    if (forceRegenerate) {
      const now = Date.now()
      const timeSinceLastRegen = now - lastForceRegenTimeRef.current
      if (timeSinceLastRegen < FORCE_REGEN_COOLDOWN_MS) {
        const waitSeconds = Math.ceil((FORCE_REGEN_COOLDOWN_MS - timeSinceLastRegen) / 1000)
        setRemainingCooldown(waitSeconds)
        return
      }
      lastForceRegenTimeRef.current = now
    }

    setLoading(true)
    setError(null)

    try {
      const response = await fetch(`/api/analysis/${pageType}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          data,
          dateRange,
          context,
          forceRegenerate
        })
      })

      if (!response.ok) {
        const errorData = await response.json().catch(() => ({}))
        throw new Error(errorData.message || 'Failed to generate analysis')
      }

      const result = await response.json()
      setAnalysis(result.analysis)
      setHasGenerated(true)
    } catch (err: any) {
      setError(err.message || 'An error occurred')
      console.error('AI Analysis error:', err)
    } finally {
      setLoading(false)
    }
  }

  // Handle clicking on analysis items
  const handleAnalysisItemClick = (itemText: string, sectionType: string) => {
    // Dispatch event for ChatPanel
    window.dispatchEvent(
      new CustomEvent('chat-send-message', {
        detail: {
          message: buildChatPrompt(itemText, sectionType, pageType, dateRange),
          displayContent: `Explain: "${itemText}"`,
          openChat: true
        }
      })
    )
  }

  // Event delegation for click handling
  useEffect(() => {
    const container = document.querySelector(`[data-analysis-container="${pageType}"]`)
    if (!container) return

    const handleClick = (event: Event) => {
      const target = (event.target as HTMLElement).closest('[data-analysis-item]')
      if (target) {
        const itemText = target.getAttribute('data-item-text')
        const sectionType = target.getAttribute('data-section-type')
        if (itemText && sectionType) {
          handleAnalysisItemClick(itemText, sectionType)
        }
      }
    }

    container.addEventListener('click', handleClick)
    return () => container.removeEventListener('click', handleClick)
  }, [analysis, pageType])

  // Render
  return (
    <Card className={`glass-luxury-card border border-gray-200/10 ${className}`}>
      <CardHeader className="pb-2 pt-3 px-4">
        <div className="flex items-center justify-between gap-4">
          <div className="flex items-center gap-3 min-w-0 flex-1">
            <div className="p-2 rounded-lg bg-amber-500/10 flex-shrink-0">
              <Sparkles className="w-5 h-5 text-amber-500" />
            </div>
            <CardTitle className="text-lg font-bold theme-text-primary truncate">
              {config.displayName}
            </CardTitle>
          </div>
          <Button
            onClick={() => generateAnalysis(true)}
            disabled={loading || remainingCooldown > 0}
            variant="ghost"
            size="sm"
            className="h-8 px-3"
          >
            {loading ? (
              <>
                <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" />
                Analyzing...
              </>
            ) : remainingCooldown > 0 ? (
              <>
                <RefreshCw className="mr-1.5 h-3.5 w-3.5 opacity-50" />
                Wait {remainingCooldown}s
              </>
            ) : hasGenerated ? (
              <>
                <RefreshCw className="mr-1.5 h-3.5 w-3.5" />
                Refresh
              </>
            ) : (
              <>
                <Sparkles className="mr-1.5 h-3.5 w-3.5" />
                Generate
              </>
            )}
          </Button>
        </div>
      </CardHeader>

      <CardContent className="px-4 pb-4 pt-2" data-analysis-container={pageType}>
        {error ? (
          <div className="flex flex-col items-center justify-center py-8 text-center">
            <AlertCircle className="w-8 h-8 text-red-500 mb-2" />
            <p className="text-sm theme-text-secondary mb-2">{error}</p>
            <Button onClick={() => generateAnalysis(false)} size="sm" variant="outline">
              Try Again
            </Button>
          </div>
        ) : loading ? (
          <div className="flex flex-col items-center justify-center py-12">
            <Loader2 className="w-8 h-8 animate-spin text-amber-500 mb-2" />
            <p className="text-sm theme-text-primary">Analyzing {config.displayName}...</p>
          </div>
        ) : analysis ? (
          <div
            className="ai-analysis-content"
            dangerouslySetInnerHTML={{
              __html: DOMPurify.sanitize(formatAnalysis(analysis), {
                ADD_ATTR: ['data-analysis-item', 'data-item-text', 'data-section-type', 'title'],
                ADD_TAGS: ['svg', 'path', 'strong'],
                ALLOW_DATA_ATTR: true
              })
            }}
          />
        ) : (
          <div className="flex flex-col items-center justify-center py-12 text-center">
            <div className="p-3 rounded-full bg-amber-500/5 mb-4">
              <Sparkles className="w-10 h-10 text-amber-500/40" />
            </div>
            <h3 className="text-base font-semibold theme-text-primary mb-2">Ready to Analyze</h3>
            <p className="text-sm theme-text-secondary max-w-md">
              Click "Generate Analysis" to get AI-powered insights
            </p>
          </div>
        )}
      </CardContent>
    </Card>
  )
}

// Helper function to build chat prompts
function buildChatPrompt(
  itemText: string,
  sectionType: string,
  pageType: string,
  dateRange: { start: string; end: string }
): string {
  const contextMap: Record<string, string> = {
    strategic: `The user clicked on this strategic insight from the ${pageType} analysis: "${itemText}"

Please provide a detailed explanation in well-formatted markdown. Include:
1. What this insight means for the business
2. The underlying data patterns that support this conclusion
3. Potential root causes and contributing factors
4. How to validate or investigate this further

Context: Analysis period ${dateRange.start} to ${dateRange.end}.`,

    forward: `The user clicked on this forward-looking prediction from the ${pageType} analysis: "${itemText}"

Please provide a detailed analysis in well-formatted markdown. Include:
1. The key assumptions behind this prediction
2. What factors could influence the outcome
3. Early indicators to watch for
4. Recommended actions to improve the predicted outcome

Context: Analysis period ${dateRange.start} to ${dateRange.end}.`,

    actions: `The user clicked on this recommended action from the ${pageType} analysis: "${itemText}"

Please provide a detailed action plan in well-formatted markdown. Include:
1. Step-by-step implementation guide
2. Resources and timeline needed
3. Expected impact and how to measure success
4. Potential risks and mitigation strategies

Context: Analysis period ${dateRange.start} to ${dateRange.end}.`
  }

  return contextMap[sectionType] || contextMap.strategic
}
```

```typescript
/**
 * Format AI analysis text into styled HTML
 * Reuses existing formatting logic from AIAnalysisCard
 */
export function formatAnalysis(text: string): string {
  // Process markdown bold syntax
  const processBold = (text: string, colorClass: string = 'text-amber-600'): string => {
    return text.replace(
      /\*\*([^*]+)\*\*/g,
      `<strong class="font-semibold ${colorClass}">$1</strong>`
    )
  }

  let formatted = text
    // Headers
    .replace(
      /^## STRATEGIC INSIGHTS$/gm,
      '<h3 class="text-sm font-bold uppercase tracking-wider theme-text-primary border-b border-blue-500/30 pb-1 mb-3">Strategic Insights</h3>'
    )
    .replace(
      /^## FORWARD-LOOKING ANALYSIS$/gm,
      '<h3 class="text-sm font-bold uppercase tracking-wider text-purple-600 border-b border-purple-500/30 pb-1 mb-3">Forward-Looking Analysis</h3>'
```

**Purpose:** Single endpoint that handles all page types dynamically.

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { createLLM } from '@/lib/llm-factory'
import { tokenTracker } from '@/lib/ai/utils/globalTokenTracker'
import { withActiveProvider } from '@/lib/providers/withActiveProvider'
import { logger } from '@/lib/logger'
import { buildSystemPrompt, buildUserPrompt } from '@/lib/ai/analysis/prompts'
import { saveAnalysis, getLatestAnalysis } from '@/lib/services/aiAnalysisStorage'
import {
  validateListFormat,
  logValidationFailure,
  logFinalValidationStatus,
} from '@/lib/ai/validation/responseValidator'
import { AI_ANALYSIS_CONFIG, isValidPageType } from '@/components/ai-analysis/ai-analysis.config'
import type { AnalysisRequest, AnalysisResponse } from '@/components/ai-analysis/types'

export const POST = withActiveProvider(
  async (
    request: NextRequest,
    { provider, organizationId, userId, providerId },
    { params }: { params: { pageType: string } }
  ) => {
    try {
      const { pageType } = params

      // Validate page type
      if (!isValidPageType(pageType)) {
        return NextResponse.json({ error: `Invalid page type: ${pageType}` }, { status: 400 })
      }

      const config = AI_ANALYSIS_CONFIG[pageType]

      logger.workflow(`${pageType}-analysis`, 'request_started', { userId, organizationId })

      // Parse request body
      const body: AnalysisRequest = await request.json()
      const { data, dateRange, context, forceRegenerate } = body

      // Validate required fields
      if (!data || typeof data !== 'object') {
        return NextResponse.json({ error: 'Invalid request: data is required' }, { status: 400 })
      }

      if (!dateRange || !dateRange.start || !dateRange.end) {
        return NextResponse.json(
          { error: 'Invalid request: dateRange with start and end is required' },
          { status: 400 }
        )
      }

      // Validate date format
      const dateRegex = /^\d{4}-\d{2}-\d{2}$/
      if (!dateRegex.test(dateRange.start) || !dateRegex.test(dateRange.end)) {
        return NextResponse.json({ error: 'Invalid date format: use YYYY-MM-DD' }, { status: 400 })
      }

      // Check cache (unless force regenerate)
      if (!forceRegenerate) {
        logger.workflow(`${pageType}-analysis`, 'checking_cache', { organizationId })
        const cached = await getLatestAnalysis(organizationId, config.analysisType)

        if (cached) {
          logger.workflow(`${pageType}-analysis`, 'cache_hit', {
            organizationId,
            analysisId: cached.analysis_id,
            remainingMinutes: Math.floor((cached.expires_at - Date.now()) / 60000),
          })

          return NextResponse.json({
            analysis: cached.analysis_text,
            tokensUsed: 0,
            cached: true,
            cacheExpiresAt: cached.expires_at,
          })
        }
      } else {
        logger.workflow(`${pageType}-analysis`, 'force_regenerate', { organizationId })
      }

      logger.workflow(`${pageType}-analysis`, 'cache_miss', { organizationId })

      // Start token tracking
      const sessionId = tokenTracker.startSession(`${pageType}_analysis`)

      // Build prompts
      const systemPrompt = buildSystemPrompt(pageType, config)
      const userPrompt = buildUserPrompt(pageType, data, dateRange, context, config)

      // Retry configuration
      const MAX_RETRIES = 3
      const RETRY_DELAYS = [500, 1000, 2000]
      let analysisContent = ''
      let validationResult
      let attempt = 0

      // Retry loop
      for (attempt = 0; attempt <= MAX_RETRIES; attempt++) {
        if (attempt > 0) {
          logger.workflow(`${pageType}-analysis`, 'retrying', {
            organizationId,
            attempt: attempt + 1,
            maxAttempts: MAX_RETRIES + 1,
            delay: RETRY_DELAYS[attempt - 1],
          })
          await new Promise((resolve) => setTimeout(resolve, RETRY_DELAYS[attempt - 1]))
        }

        // Create LLM instance
        const llm = createLLM({
          provider: 'groq',
          model: 'openai/gpt-oss-120b',
          component: `${pageType}-analysis`,
          sessionId,
          temperature: 0.7,
          maxTokens: 2000,
        })

        // Generate analysis
        const response = await llm.invoke([
          { role: 'system', content: systemPrompt },
          { role: 'user', content: userPrompt },
        ])

        analysisContent =
          typeof response === 'string'
            ? response
            : typeof response.content === 'string'
              ? response.content
              : response.text || ''

        // Validate format
        validationResult = validateListFormat(analysisContent)

        if (validationResult.isValid) {
          logger.workflow(`${pageType}-analysis`, 'validation_success', {
            organizationId,
            attempt: attempt + 1,
          })
          break
        } else {
          logValidationFailure(attempt + 1, organizationId, validationResult, analysisContent)

          if (attempt === MAX_RETRIES) {
            logger.workflow(`${pageType}-analysis`, 'max_retries_reached', {
              organizationId,
              totalAttempts: attempt + 1,
              failedSections: validationResult.failedSections,
            })
          }
        }
      }

      // End token tracking
      tokenTracker.endSession(sessionId)

      // Log final validation status
      logFinalValidationStatus(organizationId, attempt + 1, validationResult!)

      // Save to cache
      logger.workflow(`${pageType}-analysis`, 'saving_to_cache', { organizationId })
      const savedAnalysis = await saveAnalysis(
        organizationId,
        userId,
        config.analysisType,
        analysisContent,
        dateRange,
        { data, context },
        providerId,
        'llama-3.3-70b-groq',
        config.cacheTTL
      )

      logger.workflow(`${pageType}-analysis`, 'analysis_saved', {
        organizationId,
        analysisId: savedAnalysis.analysis_id,
      })

      return NextResponse.json({
        analysis: analysisContent,
        tokensUsed: tokenTracker.getSessionTotal(sessionId),
        cached: false,
        cacheExpiresAt: savedAnalysis.expires_at,
        validation: {
          passed: validationResult!.isValid,
          attempts: attempt + 1,
          failedSections: validationResult!.failedSections,
        },
      })
    } catch (error: any) {
      const errorMessage = error?.message || 'Unknown error'
      const errorName = error?.name || 'Error'

      logger.error(`Error generating ${params.pageType} analysis`, {
        error,
        errorMessage,
        errorName,
        userId,
        organizationId,
      })

      // Error handling (same as before)
      if (
        errorMessage.includes('timeout') ||
        errorMessage.includes('ETIMEDOUT') ||
        errorMessage.includes('rate limit')
      ) {
        return NextResponse.json(
          {
            error: 'AI service temporarily unavailable',
            message: 'Please try again in a moment.',
            retry: true,
          },
          { status: 503 }
        )
      }

      if (errorMessage.includes('DynamoDB') || errorMessage.includes('ResourceNotFoundException')) {
        return NextResponse.json(
          {
            error: 'Storage error',
            message: 'Unable to save analysis. Please try again.',
            cached: false,
          },
          { status: 500 }
        )
      }

      return NextResponse.json(
        {
          error: 'An unexpected error occurred',
          message: 'Please try again or contact support.',
        },
        { status: 500 }
      )
    }
  }
)
```

```typescript
/**
 * Determine if analyzing historical data and provide context
 */
function getDateContext(dateRange: DateRange): string {
  const endDate = new Date(dateRange.end)
  const now = new Date()
  const daysDiff = Math.floor((now.getTime() - endDate.getTime()) / (1000 * 60 * 60 * 24))

  if (daysDiff > 60) {
    // More than 2 months old - definitely historical
    const monthsAgo = Math.floor(daysDiff / 30)
    return `IMPORTANT: You are analyzing HISTORICAL data from ${monthsAgo} months ago (${dateRange.start} to ${dateRange.end}). Do not reference current date or make assumptions about "today". Frame all insights in past tense where appropriate.`
  } else if (daysDiff > 7) {
    // 1 week to 2 months - recent historical
    return `NOTE: This data is from ${daysDiff} days ago (${dateRange.start} to ${dateRange.end}). Consider this when making predictions.`
  } else {
    // Current period
    return `You are analyzing CURRENT data for period ${dateRange.start} to ${dateRange.end}.`
  }
}

function buildPnLPrompt(
  data: PageAnalysisData,
  dateRange: DateRange,
  dateContext: string,
  context?: AnalysisContext
): string {
  const { metrics, previousPeriod, twoPeriodAgo } = data

  // Format helpers
  const fmt = (value: number | undefined) => {
    if (!value && value !== 0) return '$0'
    const absValue = Math.abs(value)
    if (absValue >= 1000000) return `$${(value / 1000000).toFixed(1)}M`
    if (absValue >= 1000) return `$${(value / 1000).toFixed(0)}K`
    return `$${value.toFixed(0)}`
  }

  const pct = (value: number | undefined) => {
    if (!value && value !== 0) return '0%'
    return `${(value * 100).toFixed(1)}%`
  }

  // Extract metrics
  const revenue = metrics.totalRevenue || 0
  const expenses = metrics.totalExpenses || 0
  const netIncome = metrics.netIncome || 0
  const grossMargin = metrics.grossMargin || 0
  const operatingMargin = metrics.operatingMargin || 0

  // Calculate trends if previous data available
  let trendsText = ''
  if (previousPeriod) {
    const prevRevenue = previousPeriod.metrics.totalRevenue || revenue * 0.9
    const prevExpenses = previousPeriod.metrics.totalExpenses || expenses * 0.95
    const twoPeriodRev = twoPeriodAgo?.metrics.totalRevenue

    const revenueTrend = calculateTrend(revenue, prevRevenue, twoPeriodRev)
    const expenseTrend = calculateTrend(expenses, prevExpenses)

    trendsText = `
TRENDS:
Revenue: ${revenueTrend.changeRate > 0 ? '+' : ''}${revenueTrend.changeRate.toFixed(1)}% change (${revenueTrend.trend})
Expenses: ${expenseTrend.changeRate > 0 ? '+' : ''}${expenseTrend.changeRate.toFixed(1)}% change (${expenseTrend.trend})
`
  }

  return `${dateContext}

Analyze this P&L data for ${dateRange.start} to ${dateRange.end}:

CORE METRICS:
Revenue: ${fmt(revenue)}
Expenses: ${fmt(expenses)}
Net Income: ${fmt(netIncome)}
Gross Margin: ${pct(grossMargin)}
Operating Margin: ${pct(operatingMargin)}
${trendsText}

Provide strategic P&L analysis focusing on revenue drivers, expense efficiency, margin trends, and profitability optimization. Follow the specified 3-section structure.`

Analyze this Cash Flow data for ${dateRange.start} to ${dateRange.end}:

LIQUIDITY METRICS:
Operating Cash Flow: ${fmt(ocf)}

${runwayText}

WORKING CAPITAL:
DSO: ${dso} days
DPO: ${dpo} days
DIO: ${dio} days
Cash Conversion Cycle: ${cashCycle.cycle} days (${cashCycle.efficiency})
```

```typescript
// src/lib/ai/analysis/rate-limiter.ts
const requestCounts = new Map<string, { count: number; resetAt: number }>()

export function checkRateLimit(
  orgId: string,
  userId: string,
  limits: { perOrg: number; perUser: number; windowMs: number }
): { allowed: boolean; retryAfter?: number } {
  const now = Date.now()

  // Check org limit
  const orgKey = `org:${orgId}`
  const orgData = requestCounts.get(orgKey)

  if (orgData) {
    if (now < orgData.resetAt) {
      if (orgData.count >= limits.perOrg) {
        return { allowed: false, retryAfter: Math.ceil((orgData.resetAt - now) / 1000) }
      }
      orgData.count++
    } else {
      requestCounts.set(orgKey, { count: 1, resetAt: now + limits.windowMs })
    }
  } else {
    requestCounts.set(orgKey, { count: 1, resetAt: now + limits.windowMs })
  }

  // Check user limit (similar logic)
  const userKey = `user:${userId}`
  const userData = requestCounts.get(userKey)

  if (userData) {
    if (now < userData.resetAt) {
      if (userData.count >= limits.perUser) {
        return { allowed: false, retryAfter: Math.ceil((userData.resetAt - now) / 1000) }
      }
      userData.count++
    } else {
      requestCounts.set(userKey, { count: 1, resetAt: now + limits.windowMs })
    }
  } else {
    requestCounts.set(userKey, { count: 1, resetAt: now + limits.windowMs })
  }

  return { allowed: true }
}

// Cleanup old entries periodically
setInterval(() => {
  const now = Date.now()
  for (const [key, data] of requestCounts.entries()) {
    if (now >= data.resetAt) {
      requestCounts.delete(key)
    }
  }
}, 60000) // Every minute
```
